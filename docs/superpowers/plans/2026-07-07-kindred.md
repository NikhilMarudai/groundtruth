# Kindred Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a graph-aware agentic app that turns a one-line product idea into a frontier-weighted, cited reference architecture drawn from real companies — deployed live on `butterbase.dev`.

**Architecture:** A Neo4j graph of real companies↔tools (built by a RocketRide curation pipeline) is queried by a RocketRide recommendation pipeline (idea → similar companies → ranked stack → synthesized architecture). A Butterbase backend (auth + Postgres + Stripe subscription) gates the synthesis behind Pro and calls the pipeline; a React frontend renders the graph + architecture card. Cognee (memory) and Daytona (scaffold) are bonus tracks.

**Tech Stack:** Neo4j Aura + GDS · Python (`neo4j`, `graphdatascience`, `anthropic`) · RocketRide Cloud pipelines · Butterbase (functions/TS, Postgres, billing, AI gateway) · React + Vite + TypeScript + `react-force-graph-2d` · Cognee OSS · Daytona SDK.

## Global Constraints

- **Mandatory + load-bearing:** Butterbase (DB **+ auth + payment** all active), Neo4j (real traversal/GDS, not KV), RocketRide **Cloud** (deployed to `cloud.rocketride.ai`, not local). Bolted-on = DQ.
- **Demo runs on seeded data, zero live external dependency** — curation is pre-run at build time; only the recommendation pipeline is called live.
- **Every `USES` edge carries a `Source`** (citation). Framed in-product as "publicly-observed stack."
- **`CURRENT_YEAR = 2026`** (hardcode; no runtime date calls in pipeline nodes).
- **Secrets via env only** — never commit keys. `.env` is gitignored; `.env.example` lists names.
- **Submission:** promo `ENJOY0707`, slug `HackwithBay-0707`.
- **Frontier ranking is the differentiator** — recency × stage × adoption-slope, exact formula in Task A3.

```python
# graph/etl.py
import json
LOAD = """
UNWIND $rows AS r
MERGE (c:Company {name:r.name}) SET c.founded=r.founded, c.stage=r.stage
MERGE (dom:Domain {name:r.domain}) MERGE (c)-[:IN_DOMAIN]->(dom)
FOREACH (a IN r.context_attrs | MERGE (at:Attribute {name:a}) MERGE (c)-[:HAS_CONTEXT {weight:1.0}]->(at))
WITH c, r
UNWIND r.tools AS t
  MERGE (tool:Tool {name:t.name}) SET tool.layer=t.layer
  MERGE (c)-[u:USES]->(tool) SET u.since=t.since, u.confidence=1.0
  MERGE (src:Source {url:t.source.url}) SET src.title=t.source.title
  MERGE (c)-[:CITES {tool:t.name}]->(src)
  FOREACH (k IN t.satisfies | MERGE (kc:Constraint {name:k}) MERGE (tool)-[:SATISFIES]->(kc))
  FOREACH (k IN t.violates  | MERGE (kc:Constraint {name:k}) MERGE (tool)-[:VIOLATES]->(kc))
"""
def load(driver, path):
    rows = json.load(open(path))
    driver.execute_query(LOAD, rows=rows)
```

```python
# graph/algorithms.py
from graphdatascience import GraphDataScience
import os
def gds(): return GraphDataScience(os.environ["NEO4J_URI"], auth=(os.environ["NEO4J_USER"], os.environ["NEO4J_PASSWORD"]))
def build(driver):
    g = gds()
    driver.execute_query("MATCH ()-[r:SIMILAR_TO]->() DELETE r")
    driver.execute_query("MATCH ()-[r:CO_OCCURS]->() DELETE r")
    # co-occurrence of tools across companies
    driver.execute_query("""
      MATCH (a:Tool)<-[:USES]-(c:Company)-[:USES]->(b:Tool) WHERE a.name<b.name
      WITH a,b,count(c) AS n MERGE (a)-[r:CO_OCCURS]->(b) SET r.count=n """)
    # company similarity over shared context/domain (Jaccard via GDS nodeSimilarity on a projection)
    G,_ = g.graph.project("csim", ["Company","Attribute","Domain"],
        {"HAS_CONTEXT":{"orientation":"UNDIRECTED"},"IN_DOMAIN":{"orientation":"UNDIRECTED"}})
    g.nodeSimilarity.write(G, writeRelationshipType="SIMILAR_TO", writeProperty="score", topK=15)
    g.louvain.write(G, writeProperty="community")
    G.drop()
    # materialize archetypes from community ids
    driver.execute_query("""
      MATCH (c:Company) WHERE c.community IS NOT NULL
      MERGE (sa:StackArchetype {name:'archetype-'+toString(c.community)}) MERGE (c)-[:MEMBER_OF]->(sa)""")
```

- [ ] **Step 3: Implement `graph/queries.py`** (frontier formula per Global Constraints).

```python
# graph/queries.py
CURRENT_YEAR = 2026
STAGE_W = {"seed":1.0,"early":0.9,"growth":0.6,"public":0.4,"legacy":0.35}
def _recency_w(since):
    since = since or 2015
    return max(0.2, min(1.0, (since-2015)/(CURRENT_YEAR-2015)))

def match_companies(driver, idea_profile, k=8):
    q="""
    MATCH (c:Company)-[:HAS_CONTEXT]->(a:Attribute) WHERE a.name IN $attrs
    WITH c, count(a) AS overlap
    OPTIONAL MATCH (c)-[:IN_DOMAIN]->(d:Domain) WHERE d.name=$domain
    WITH c, overlap + CASE WHEN d IS NULL THEN 0 ELSE 2 END AS sim
    ORDER BY sim DESC LIMIT $k
    MATCH (c)-[:USES]->(t:Tool)
    RETURN c.name AS name, c.stage AS stage, c.founded AS founded,
           toFloat(sim) AS similarity, collect(t.name) AS tools ORDER BY similarity DESC"""
    rows=driver.execute_query(q, attrs=idea_profile["context_attrs"], domain=idea_profile["domain"], k=k)[0]
    return [dict(r) for r in rows]

def rank_tools(driver, idea_profile, constraints, k_companies=8):
    cos=match_companies(driver, idea_profile, k_companies)
    names=[c["name"] for c in cos]
    q="""
    MATCH (c:Company)-[u:USES]->(t:Tool) WHERE c.name IN $names
    WHERE NOT EXISTS { MATCH (t)-[:VIOLATES]->(k:Constraint) WHERE k.name IN $cons }
    OPTIONAL MATCH (c)-[cit:CITES {tool:t.name}]->(s:Source)
    RETURN t.name AS tool, t.layer AS layer, c.name AS company, c.stage AS stage,
           u.since AS since, c.founded AS founded,
           collect(DISTINCT {title:s.title,url:s.url,company:c.name})[0..3] AS cites"""
    rows=[dict(r) for r in driver.execute_query(q, names=names, cons=constraints)[0]]
    agg={}
    for r in rows:
        base=STAGE_W.get(r["stage"],0.5)*_recency_w(r["since"])
        e=agg.setdefault(r["tool"],{"tool":r["tool"],"layer":r["layer"],"score":0.0,
                                    "recent":0,"total":0,"citations":[]})
        e["score"]+=base; e["total"]+=1
        if (r["founded"] or 0) >= CURRENT_YEAR-3: e["recent"]+=1
        e["citations"]+=[c for c in r["cites"] if c["url"]]
    out=[]
    for e in agg.values():
        slope=e["recent"]/e["total"] if e["total"] else 0
        out.append({"tool":e["tool"],"layer":e["layer"],
                    "frontier_score":round(e["score"]*(1+0.5*slope),3),
                    "rising":slope>=0.5,
                    "citations":e["citations"][:3]})
    return sorted(out,key=lambda x:-x["frontier_score"])

def stack_archetype(driver, company_names):
    q="""MATCH (c:Company)-[:MEMBER_OF]->(sa:StackArchetype) WHERE c.name IN $names
         RETURN sa.name AS a, count(*) AS n ORDER BY n DESC LIMIT 1"""
    r=driver.execute_query(q, names=company_names)[0]
    return r[0]["a"] if r else "mixed"

def explain_path(driver, idea_profile, company_names):
    q="""MATCH (c:Company)-[:USES]->(t:Tool) WHERE c.name IN $names
```

```python
import json, os
from tavily import TavilyClient
import anthropic
_tv=TavilyClient(os.environ["TAVILY_API_KEY"]); _an=anthropic.Anthropic()
def _search(q): return _tv.search(q, max_results=4)["results"]
EXTRACT_PROMPT = """You are a precise tech-stack extractor. From the SEARCH RESULTS about "{name}",
return STRICT JSON per this schema. Only include a tool if a result explicitly evidences it; attach the
exact source url+title. If unsure, omit it. Never invent. Schema:
{{"name","founded":int,"stage":"seed|early|growth|public|legacy","domain","context_attrs":[..],
 "tools":[{{"name","layer":"db|backend|frontend|infra|ai|auth|other","since":int,
   "source":{{"url","title"}},"satisfies":[..],"violates":[..]}}]}}
SEARCH RESULTS:\n{ctx}"""
def _llm_extract(name, ctx):
    msg=_an.messages.create(model="claude-sonnet-5",max_tokens=1500,
        messages=[{"role":"user","content":EXTRACT_PROMPT.format(name=name,ctx=ctx)}])
    return json.loads(msg.content[0].text)
def curate_one(name):
    res=_search(f"{name} tech stack engineering blog what database they use")
    ctx="\n\n".join(f"{r['title']} ({r['url']}): {r.get('content','')}" for r in res)
    return _llm_extract(name, ctx)
def run(list_path, out_path):
    names=json.load(open(list_path)); out=[]
    for n in names:
        try:
            o=curate_one(n); out.append(o); print(f"{n}: tools={len(o.get('tools',[]))}")
        except Exception as e: print(f"{n}: FAILED {e}")
    json.dump(out, open(out_path,"w"), indent=2); print(f"wrote {len(out)} -> {out_path}")
```
